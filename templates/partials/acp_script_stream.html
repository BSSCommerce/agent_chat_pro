async function sendStream(thread, queryText, imageUrls = []) { const response = await fetch("/api/agent/query/stream", { method: "POST", credentials: "same-origin", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ query: queryText, agent_alias: thread.agent.alias, thread_id: thread.external_thread_id, image_urls: normalizeMessageImageUrls(imageUrls), }), }); if (!response.ok || !response.body) { const data = await response.json().catch(() => ({})); throw new Error(data.detail || `Agent request failed: ${response.status}`); } const turn = createStreamingTurn(); const reader = response.body.getReader(); const decoder = new TextDecoder(); let buffer = ""; let finalAnswer = ""; // Latest model text not yet known to be final. When a newer message or the // authoritative `answer` arrives, the previous one is "working", not the answer. let pendingMessage = ""; let interrupted = false; function flushPendingToWorking() { if (pendingMessage) { turn.commitMessage(pendingMessage); pendingMessage = ""; } } function applyEvent(eventName, dataText) { const eventType = (eventName || "").trim() || "message"; if (!dataText && eventType !== "done") return; let payload = {}; try { payload = JSON.parse(dataText); } catch { payload = {}; } if (eventType === "error") { throw new Error(payload.detail || "Stream error"); } if (eventType === "start" && payload.thread_id && thread?.external_thread_id) { const sid = String(payload.thread_id).trim(); if (sid && sid !== String(thread.external_thread_id).trim()) { setStatus("Stream thread id mismatch — check agent runtime.", true); } } if (eventType === "step" && payload.message) { if ((payload.kind || "").trim() === "activity") { // Live ACP processing timeline → working panel (cumulative render). turn.pushActivity(payload.message); } else { // Model text: hold as the candidate answer; the prior candidate is // now a superseded step and moves into the working timeline. flushPendingToWorking(); pendingMessage = payload.message; turn.setAnswer(pendingMessage); } setStatus("Receiving response..."); } if (eventType === "answer" && payload.message) { // Authoritative final answer — never part of "working". pendingMessage = ""; finalAnswer = payload.message; turn.setAnswer(finalAnswer); setStatus("Receiving response..."); } if (eventType === "interrupt" && Array.isArray(payload.interrupts) && payload.interrupts.length) { interrupted = true; pendingMessage = ""; finalAnswer = "Run paused and requires approval."; turn.setAnswer(finalAnswer); renderApprovalPanel(payload.interrupts); setStatus("Waiting for approval", true); } if (eventType === "done") { if (payload.status === "interrupted") { interrupted = true; return; } hideApprovalPanel(); } } while (true) { const { done, value } = await reader.read(); if (done) break; buffer += decoder.decode(value, { stream: true }).replace(/\r\n/g, "\n"); let boundary = buffer.indexOf("\n\n"); while (boundary !== -1) { const rawEvent = buffer.slice(0, boundary); buffer = buffer.slice(boundary + 2); let eventName = ""; const dataLines = []; for (const rawLine of rawEvent.split("\n")) { const line = rawLine.trimEnd(); if (line.startsWith("event:")) { eventName = line.slice(6).trim(); } else if (line.startsWith("data:")) { dataLines.push(line.slice(5).trimStart()); } } applyEvent(eventName, dataLines.join("\n")); boundary = buffer.indexOf("\n\n"); } } if (!finalAnswer) { // No explicit answer: fall back to the last pending message, else a notice. finalAnswer = pendingMessage || "(stream finished without visible answer)"; turn.setAnswer(finalAnswer); } turn.finish(); // Cap the saved log well under the API limit so a very long run never blocks // persistence of the assistant answer itself. const ACP_ACTIVITY_LOG_MAX = 180000; let activityMarkdown = turn.activityMarkdown(); if (activityMarkdown.length > ACP_ACTIVITY_LOG_MAX) { activityMarkdown = activityMarkdown.slice(0, ACP_ACTIVITY_LOG_MAX) + "\n\n…(working log truncated)"; } const activityLog = activityMarkdown ? JSON.stringify({ v: 1, duration_ms: turn.durationMs(), markdown: activityMarkdown }) : null; return { answer: finalAnswer, status: interrupted ? "interrupted" : "completed", activityLog, }; } async function sendMessage() { const rawUserQuery = (messageInput.value || "").trim(); const outgoingImages = [...selectedImageUrls]; if (!rawUserQuery && !outgoingImages.length) { setStatus("Enter a message or attach at least one image.", true); messageInput.focus(); return; } const agent = selectedAgent(); if (!agent) { setStatus("Please select an agent.", true); agentSelect.focus(); return; } const priorSnapshot = [...(state.loadedMessages || [])]; let agentQueryText = rawUserQuery; if (state.currentThread?.fork_bootstrap_pending && priorSnapshot.length) { agentQueryText = buildForkAugmentedQuery(priorSnapshot, rawUserQuery); } setBusy(true); setStatus("Preparing thread..."); try { const thread = await createThreadIfNeeded(); appendBubbleWithActions({ id: null, role: "user", content: rawUserQuery, image_urls: outgoingImages, status: "completed", }); messageInput.value = ""; clearComposerImages(); await saveMessage(thread.id, "user", rawUserQuery, "completed", outgoingImages); await reloadCurrentThreadMessages(); setStatus("Streaming response..."); const result = await sendStream(state.currentThread, agentQueryText, outgoingImages); await saveMessage( state.currentThread.id, "assistant", result.answer, result.status, [], result.activityLog, ); if (state.currentThread?.fork_bootstrap_pending) { await apiFetch(`/agent-chat-pro/api/threads/${state.currentThread.id}`, { method: "PATCH", body: JSON.stringify({ fork_bootstrap_pending: false }), }); state.currentThread = { ...state.currentThread, fork_bootstrap_pending: false }; } await reloadCurrentThreadMessages(); await refreshThreads(); setStatus(result.status === "interrupted" ? "Run paused for approval" : "Done"); } catch (error) { setStatus(error.message || "Unexpected error", true); appendBubble("assistant", `Error: ${error.message || "Unexpected error"}`, "error"); if (state.currentThread) { try { await saveMessage(state.currentThread.id, "assistant", `Error: ${error.message || "Unexpected error"}`, "error"); await reloadCurrentThreadMessages(); } catch { // The visible error is enough if persistence also fails. } } } finally { setBusy(false); } }
